cmake_minimum_required(VERSION 3.22)
project(rankedtools_https_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(RANKEDTOOLS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(COMPILE_ID rankedtools_https_smoke)
add_executable(${COMPILE_ID} ../https_smoke.cpp)
include("${RANKEDTOOLS_ROOT}/cmake/https.cmake")
add_executable(rankedtools_https_threading ../https_threading_tests.cpp)
target_link_libraries(rankedtools_https_threading PRIVATE mbedcrypto)
foreach(target ${COMPILE_ID} rankedtools_https_threading)
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHsc /utf-8 /Gy)
        target_link_options(${target} PRIVATE /OPT:REF)
    else()
        target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
        target_link_options(${target} PRIVATE -static -Wl,--gc-sections)
    endif()
    target_include_directories(${target} PRIVATE
        "${RANKEDTOOLS_ROOT}/include" "${RANKEDTOOLS_ROOT}/pcvr/native/compat"
        "${RANKEDTOOLS_ROOT}/extern/includes"
        "${RANKEDTOOLS_ROOT}/extern/includes/beatsaber-hook/shared/rapidjson/include")
    target_compile_definitions(${target} PRIVATE CURL_STATICLIB)
endforeach()
